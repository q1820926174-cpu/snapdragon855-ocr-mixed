cmake_minimum_required(VERSION 3.16)
project(sm8150_ocr_mixed LANGUAGES C CXX)
find_package(ncnn CONFIG REQUIRED)
add_executable(pnnx-gpu-runner src/gpu.cpp)
target_compile_features(pnnx-gpu-runner PRIVATE cxx_std_14)
target_link_libraries(pnnx-gpu-runner PRIVATE ncnn)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_compile_options(pnnx-gpu-runner PRIVATE -fno-tree-loop-vectorize -fno-tree-slp-vectorize)
endif()
option(BUILD_MIXED "Build the older SNPE-dependent mixed NPU/GPU executors" OFF)
if(BUILD_MIXED)
set(SNPE_ROOT "$ENV{SNPE_ROOT}" CACHE PATH "User-provided SNPE 1.59 SDK")
set(SNPE_LIBRARY "${SNPE_ROOT}/lib/aarch64-oe-linux-gcc9.3/libSNPE.so" CACHE FILEPATH "User-provided aarch64 SNPE library")
if(NOT EXISTS "${SNPE_ROOT}/include/zdl/SNPE/SNPE.hpp" OR NOT EXISTS "${SNPE_LIBRARY}")
  message(FATAL_ERROR "Set SNPE_ROOT and SNPE_LIBRARY to your authorized SDK/runtime")
endif()
foreach(profile IN ITEMS quality fast)
  if(profile STREQUAL "quality")
    set(target mixed-refined-fp32-bench)
  else()
    set(target mixed-packed-bench)
  endif()
  add_executable(${target} src/${profile}.cpp)
  target_compile_features(${target} PRIVATE cxx_std_14)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(${target} PRIVATE -fno-tree-loop-vectorize -fno-tree-slp-vectorize)
  endif()
  target_include_directories(${target} PRIVATE "${SNPE_ROOT}/include/zdl")
  target_link_libraries(${target} PRIVATE ncnn "${SNPE_LIBRARY}" dl)
endforeach()
add_library(npu-uncached-ion SHARED runtime/npu-uncached-ion.c)
target_compile_features(npu-uncached-ion PRIVATE c_std_11)
endif()
